# requirements.txt
# 由项目导入扫描生成（router.py / query_student.py / init_db.py / employee_api.py）
#
# 安装：
#     pip install -r requirements.txt
#
# 说明：
#   - 项目唯一的第三方依赖是 pymongo（用于连接 MongoDB）。
#   - router.py 的 DeepSeek / MyMemory 调用使用标准库 urllib，
#     无需 requests。
#   - 其余导入（json / os / re / socket / subprocess / sys / urllib）均为
#     Python 标准库，不写入本文件。
# ---------------------------------------------------------------------------

# MongoDB 官方 Python 驱动
# 4.6 起内置对 MongoDB 5.0+ 的完整支持；>=4.9 才正式支持 Python 3.13/3.14。
# 若本机解释器是 3.14，请勿降到 4.9 以下，否则可能装不上或运行报错。
pymongo>=4.9,<5.0


# ---------------------------------------------------------------------------
# RAG 模块依赖（rag_demo.py 使用）
#
# 安装：
#     pip install -r requirements.txt
#
# 说明：
#   - 本模块的向量化与生成全部走云端 HTTP 接口（智谱 Embedding API +
#     智谱 GLM-4-Flash / 本地 Ollama 生成），不引入 HuggingFace 本地模型，
#     因此不会安装 torch / sentence-transformers，不额外占用磁盘。
#   - langchain 提供 RecursiveCharacterTextSplitter 文本切分器；
#     langchain-community 提供 Chroma 向量库封装；
#     langchain-openai 用于调用所有 OpenAI 兼容端点
#     （智谱 https://open.bigmodel.cn/api/paas/v4 与本地 Ollama
#      http://localhost:11434/v1），因此无需安装各厂商的独立 SDK。
#   - 注意：若本机解释器为 Python 3.14，需要 chromadb>=1.5.4；
#     早期 chromadb 依赖 onnxruntime，3.14 下没有 wheel，pip 会尝试源码
#     编译并失败。若安装报错，建议为 RAG 模块单独建 Python 3.11/3.12
#     虚拟环境，再执行 pip install -r requirements.txt。
# ---------------------------------------------------------------------------

# LangChain 核心与集成
langchain
langchain-community
langchain-openai
langgraph

# 本地向量数据库
chromadb

python-dotenv
